import scipy
from scipy import stats

plt.rcParams["figure.figsize"]=(15,10) #set figure sizes
plt.rcParams["figure.dpi"]=300 #set dpi of figures (for clarity)
np.random.seed(42) #set random seed (so that it looks the same every time)

#loc and scale values can be changed here
loc=3.14
scale=0.02
data=stats.norm.rvs(loc=loc,scale=scale,size=100000)
In [3]:
#first plot (linear-linear histogram of the distribution with lines)
fig,ax=plt.subplots(1,1)
ax.hist(data,50,density=True,color="b",alpha=0.7,histtype="stepfilled") #blue histogram
plt.tick_params(labelsize=24) #set sizes of the numbers on the axes
plt.xlim([loc-5*scale,loc+5*scale]) #set limits on the x-axis
x=np.linspace(loc-5*scale,loc+5*scale,1000) #create arrays with linear spaces from -5 sigmas to +5 sigmas
ax.plot(x,stats.norm.pdf(x,loc=loc,scale=scale),linewidth=8,alpha=0.7,c="r") #red lines
plt.show() #show plot
